# 5) Build contiguous runs of the same CURRENT_OPERATION per LOTNUMBER
def build_runs(group: pd.DataFrame) -> pd.DataFrame:
    change = (group["CURRENT_OPERATION"] != group["CURRENT_OPERATION"].shift()).cumsum()
    runs = (
        group.assign(RUN_ID=change)
             .groupby("RUN_ID", as_index=False)
             .agg(
                 LOTNUMBER=("LOTNUMBER", "first"),
                 CURRENT_OPERATION=("CURRENT_OPERATION", "first"),
                 NUMBERWAFER=("NUMBERWAFER", "first"),
                 START_TIME=("TRIGGER_DATE", "first"),
                 END_TIME=("TRIGGER_DATE", "last"),
                 COUNT=("TRIGGER_ID", "count"),
             )
    )
    return runs

    # (B) Inter-run transition: prev_op - next_op (next.start - prev.end)
